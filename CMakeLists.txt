cmake_minimum_required(VERSION 3.16)
project(mvp2005_recomp CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT EXISTS ${CMAKE_SOURCE_DIR}/generated/functable.cpp)
  execute_process(COMMAND ${Python3_EXECUTABLE} tools/recomp/ee_recomp.py WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endif()
file(GLOB GEN_SRC ${CMAKE_SOURCE_DIR}/generated/*.cpp)
file(GLOB RT_SRC ${CMAKE_SOURCE_DIR}/runtime/src/*.cpp)
add_executable(mvp2005 ${GEN_SRC} ${RT_SRC})
target_include_directories(mvp2005 PRIVATE runtime/include generated)
if(MSVC)
  target_compile_options(mvp2005 PRIVATE /bigobj /w)
else()
  target_compile_options(mvp2005 PRIVATE -fno-strict-aliasing -fwrapv -w)
  # generated code is huge: keep optimisation moderate so builds finish in reasonable time
  set_source_files_properties(${GEN_SRC} PROPERTIES COMPILE_OPTIONS "-O1")
  # precompiled VU1 blocks are the hottest code in the game: full optimisation
  file(GLOB VU1_BLOCK_SRC ${CMAKE_SOURCE_DIR}/generated/vu1_blocks_*.cpp)
  if(VU1_BLOCK_SRC)
    set_source_files_properties(${VU1_BLOCK_SRC} PROPERTIES COMPILE_OPTIONS "-O2;-fstrict-aliasing")
  endif()
endif()
# where the launcher finds the list of features (a GitHub release's manifest.json); empty: none built in
set(MVP_FEATURES_URL "" CACHE STRING "URL of the features manifest.json")
if(MVP_FEATURES_URL)
  target_compile_definitions(mvp2005 PRIVATE MVP_FEATURES_URL="${MVP_FEATURES_URL}")
endif()
if(WIN32 AND NOT MSVC)
  # one .exe without a console window, needing no MinGW runtime DLLs
  target_link_options(mvp2005 PRIVATE -mwindows -static-libgcc -static-libstdc++ -Wl,-Bstatic,--whole-archive -lwinpthread -Wl,--no-whole-archive,-Bdynamic)
endif()
if(NOT MSVC)
  set_source_files_properties(${RT_SRC} PROPERTIES COMPILE_OPTIONS "-O2")
endif()
find_package(Threads REQUIRED)
target_link_libraries(mvp2005 PRIVATE Threads::Threads)
find_package(SDL2 QUIET)
if(SDL2_FOUND)
  target_compile_definitions(mvp2005 PRIVATE MVP_SDL)
  if(TARGET SDL2::SDL2)
    target_link_libraries(mvp2005 PRIVATE SDL2::SDL2)
  else()
    target_include_directories(mvp2005 PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_libraries(mvp2005 PRIVATE ${SDL2_LIBRARIES})
  endif()
else()
  message(STATUS "SDL2 not found: building headless (no window)")
endif()

# ---- graphics APIs (OpenGL is always built with SDL2)
if(SDL2_FOUND AND APPLE)
  # Metal renderer + presenter (Objective-C++)
  enable_language(OBJCXX)
  target_sources(mvp2005 PRIVATE runtime/src/metal.mm)
  set_source_files_properties(runtime/src/metal.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
  target_link_libraries(mvp2005 PRIVATE "-framework Metal" "-framework QuartzCore" "-framework Foundation")
endif()
if(SDL2_FOUND)
  # Vulkan: headers only; the loader (vulkan-1.dll / libvulkan.so.1 / MoltenVK) is opened at run time
  find_package(Vulkan QUIET COMPONENTS glslc)
  find_path(VULKAN_HEADER_DIR vulkan/vulkan.h HINTS ${Vulkan_INCLUDE_DIRS} $ENV{VULKAN_SDK}/include)
  if(VULKAN_HEADER_DIR)
    target_compile_definitions(mvp2005 PRIVATE MVP_VULKAN)
    target_include_directories(mvp2005 PRIVATE ${VULKAN_HEADER_DIR})
    # the SPIR-V is checked in (runtime/src/vk/*.inc); with glslc it is remade when a shader changes
    find_program(GLSLC glslc HINTS ${Vulkan_GLSLC_EXECUTABLE})
    if(GLSLC)
      file(GLOB VK_SHADERS ${CMAKE_SOURCE_DIR}/runtime/src/vk/*.vert ${CMAKE_SOURCE_DIR}/runtime/src/vk/*.frag)
      foreach(sh ${VK_SHADERS})
        add_custom_command(OUTPUT ${sh}.inc COMMAND ${GLSLC} --target-env=vulkan1.1 -mfmt=c -I${CMAKE_SOURCE_DIR}/runtime/src/vk ${sh} -o ${sh}.inc
                           DEPENDS ${sh} ${CMAKE_SOURCE_DIR}/runtime/src/vk/common.glsl)
        list(APPEND VK_INC ${sh}.inc)
      endforeach()
      add_custom_target(vk_shaders DEPENDS ${VK_INC})
      add_dependencies(mvp2005 vk_shaders)
    endif()
    message(STATUS "Vulkan renderer: on")
  endif()
endif()
